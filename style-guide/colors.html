05 · CSS Custom Properties

:root {
  /* ── Core Palette ──────────────────────── */
  --black:  #000000;
  --white:  #FFFFFF;
  --sand:   #D9C6AE;
  --blue:   #47578D;

  /* ── Text on Light ─────────────────────── */
  --text-primary:    #000000;
  --text-strong:     rgba(0, 0, 0, 0.85);
  --text-secondary:  rgba(0, 0, 0, 0.65);
  --text-muted:      rgba(0, 0, 0, 0.55);
  --text-light:      rgba(0, 0, 0, 0.42);
  --text-subtle:     rgba(0, 0, 0, 0.4);

  /* ── Text on Dark ──────────────────────── */
  --text-on-dark:           #FFFFFF;
  --text-on-dark-secondary: rgba(255, 255, 255, 0.75);
  --text-on-dark-muted:     rgba(255, 255, 255, 0.5);
  --text-on-dark-subtle:    rgba(255, 255, 255, 0.25);

  /* ── Borders on Light ──────────────────── */
  --border-light:   rgba(0, 0, 0, 0.07);
  --border-default: rgba(0, 0, 0, 0.15);
  --border-strong:  rgba(0, 0, 0, 0.25);

  /* ── Borders on Dark ───────────────────── */
  --border-on-dark:        rgba(255, 255, 255, 0.08);
  --border-on-dark-strong: rgba(255, 255, 255, 0.25);

  /* ── Surfaces ──────────────────────────── */
  --surface-nav: rgba(255, 255, 255, 0.96);
}
