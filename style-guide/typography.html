05 · CSS Custom Properties

:root {
  /* ── Font Families ─────────────────────── */
  --serif: 'Taviraj', Georgia, serif;
  --sans:  'Bricolage Grotesque', 'Helvetica Neue', sans-serif;

  /* ── Base Letter-spacing ───────────────── */
  --ls: -0.022em;

  /* ── Type Scale (fluid) ────────────────── */
  --text-hero:          clamp(34px, 5.2vw, 76px);
  --text-craft-hero:    clamp(30px, 4.8vw, 70px);
  --text-section:       clamp(30px, 4.2vw, 60px);
  --text-showroom:      clamp(30px, 3.8vw, 54px);
  --text-brand:         clamp(22px, 2.6vw, 36px);
  --text-craft-body:    clamp(17px, 1.9vw, 25px);
  --text-body-serif:    clamp(16px, 1.55vw, 21px);
  --text-subtitle:      clamp(13px, 1.4vw, 19px);
  --text-body:          clamp(13px, 1.2vw, 16px);
  --text-small:         clamp(13px, 1.1vw, 15px);

  /* ── Fixed Sizes ───────────────────────── */
  --text-nav:     12.5px;
  --text-button:  11px;
  --text-label:   10px;
}
